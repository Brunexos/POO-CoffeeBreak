package Entidades;

/** Produto medido por volume em mililitros. */
public class Bebida extends Produto {
    private Integer volumeML; // volume de uma unidade, em mililitros

    /** Cria uma bebida usando os atributos herdados de Produto. */
    public Bebida(String nome, Double preco, Integer quantidadeEstoque, Integer codigo, Integer volumeML) {
        super(nome, preco, quantidadeEstoque, codigo);
        setVolumeML(volumeML);
    }

    /** Retorna o volume da bebida. */
    public Integer getVolumeML() {
        return volumeML;
    }

    /** Altera o volume da bebida. */
    public void setVolumeML(Integer volumeML) {
        if (volumeML == null || volumeML <= 0) {
            throw new IllegalArgumentException("O volume deve ser positivo.");
        }
        this.volumeML = volumeML;
    }

    /** Identifica este produto como bebida na tabela do cardapio. */
    @Override
    public String getTipo() {
        return "Bebida";
    }

    /** Bebidas sao medidas em mililitros. */
    @Override
    public String getDetalhe() {
        return volumeML + " mL";
    }

    /** Copia os dados comuns (via super) e o volume; o Cardapio garante que novosDados e uma Bebida. */
    @Override
    public void atualizarDados(Produto novosDados) {
        super.atualizarDados(novosDados);
        setVolumeML(((Bebida) novosDados).getVolumeML());
    }

    /** Acrescenta o volume a descricao herdada de Produto. */
    @Override
    public String toString() {
        return super.toString() + "\n" + "volumeML: " + volumeML;
    }
}

package Entidades;

/** Um produto e sua quantidade dentro de um pedido. */
public class ItemPedido {
    private final Produto produto;      // produto do cardapio que foi pedido
    private final int quantidade;       // unidades pedidas desse produto
    private final double precoUnitario; // preco copiado no momento do pedido

    /** Guarda o preco no momento da inclusao para preservar o valor do pedido. */
    public ItemPedido(Produto produto, int quantidade) {
        if (produto == null || produto.getPreco() == null
                || !Double.isFinite(produto.getPreco()) || produto.getPreco() <= 0) {
            throw new IllegalArgumentException("Produto e preco devem ser validos.");
        }
        if (quantidade <= 0) {
            throw new IllegalArgumentException("A quantidade deve ser positiva.");
        }
        this.produto = produto;
        this.quantidade = quantidade;
        this.precoUnitario = produto.getPreco();
    }

    /** Retorna o produto associado ao item. */
    public Produto getProduto() {
        return produto;
    }

    /** Retorna o numero de unidades. */
    public int getQuantidade() {
        return quantidade;
    }

    /** Retorna o preco registrado quando o item foi criado. */
    public double getPrecoUnitario() {
        return precoUnitario;
    }

    /** Calcula o valor deste item sem depender de alteracoes futuras no cardapio. */
    public double calcularSubtotal() {
        return precoUnitario * quantidade;
    }
}

package Entidades;

import java.util.ArrayList;
import Enums.StatusPedido;
import Interfaces.FormaPagamento;

/**
 * Pedido composto por itens (composicao: a lista de itens pertence ao pedido)
 * e associado a uma forma de pagamento recebida de fora (injecao de dependencia).
 */
public class Pedido {
    private Integer codigo;                // identificador unico do pedido
    private ArrayList<ItemPedido> itens;   // itens que compoem o pedido
    private FormaPagamento formaPagamento; // interface: Pix, dinheiro ou cartao
    private StatusPedido status;           // etapa atual do pedido

    /** Recebe os itens e a forma de pagamento; nao cria essas dependencias. */
    public Pedido(Integer codigo, ArrayList<ItemPedido> itens, FormaPagamento formaPagamento) {
        setCodigo(codigo);
        validarItens(itens);
        setFormaPagamento(formaPagamento);
        this.itens = new ArrayList<>(itens); // copia: mudancas na lista original nao afetam o pedido
        this.status = StatusPedido.PENDENTE; // todo pedido novo comeca pendente
    }

    /** Soma os subtotais de todos os itens do pedido. */
    public double calcularTotal() {
        double total = 0;
        for (ItemPedido item : itens) {
            total += item.calcularSubtotal();
        }
        return total;
    }

    /** Exibe os itens e o total deste pedido quando ele e consultado. */
    public void exibirDetalhes() {
        System.out.println("Pedido " + codigo + " - " + status
                + " - " + formaPagamento.getNome());
        // Cabecalho e linhas com largura fixa formam a tabela de itens.
        System.out.printf("%-22s %8s %12s %12s%n",
                "Produto", "Qtd.", "Preco", "Subtotal");
        for (ItemPedido item : itens) {
            System.out.printf("%-22.22s %8d %12.2f %12.2f%n",
                    item.getProduto().getNome(), item.getQuantidade(),
                    item.getPrecoUnitario(), item.calcularSubtotal());
        }
        System.out.printf("Total: R$ %.2f%n", calcularTotal());
    }

    /**
     * Solicita o pagamento e da baixa no estoque dos produtos.
     * So avanca se o pedido estiver pendente, houver estoque e a forma de pagamento aprovar.
     */
    public boolean confirmarPagamento() {
        // Polimorfismo: pagar() executa a implementacao da forma de pagamento escolhida.
        if (status != StatusPedido.PENDENTE || !estoqueSuficiente()
                || !formaPagamento.pagar(calcularTotal())) {
            return false;
        }
        // O estoque so e retirado depois de conferir todos os itens, para nao baixar pela metade.
        for (ItemPedido item : itens) {
            item.getProduto().removerEstoque(item.getQuantidade());
        }
        status = StatusPedido.PROCESSANDO;
        return true;
    }

    /** Confere se o estoque cobre o total pedido de cada produto, mesmo repetido em varios itens. */
    private boolean estoqueSuficiente() {
        for (ItemPedido item : itens) {
            int unidadesDoProduto = 0;
            for (ItemPedido outro : itens) {
                if (outro.getProduto() == item.getProduto()) {
                    unidadesDoProduto += outro.getQuantidade();
                }
            }
            if (unidadesDoProduto > item.getProduto().getQuantidadeEstoque()) {
                return false;
            }
        }
        return true;
    }

    /** Conclui apenas um pedido cujo pagamento foi confirmado. */
    public boolean confirmarPedido() {
        if (status != StatusPedido.PROCESSANDO) {
            return false;
        }
        status = StatusPedido.CONCLUIDO;
        return true;
    }

    /** Cancela somente pedidos ainda pendentes. */
    public boolean cancelarPedido() {
        if (status != StatusPedido.PENDENTE) {
            return false;
        }
        status = StatusPedido.CANCELADO;
        return true;
    }

    /** Retorna o identificador do pedido. */
    public Integer getCodigo() {
        return codigo;
    }

    /** Define um identificador positivo. */
    public void setCodigo(Integer codigo) {
        if (codigo == null || codigo <= 0) {
            throw new IllegalArgumentException("O codigo deve ser positivo.");
        }
        this.codigo = codigo;
    }

    /** Devolve uma copia para impedir mudancas diretas na lista interna. */
    public ArrayList<ItemPedido> getItens() {
        return new ArrayList<>(itens);
    }

    /** Permite alterar os itens somente antes de confirmar o pagamento. */
    public void setItens(ArrayList<ItemPedido> itens) {
        if (status != StatusPedido.PENDENTE) {
            throw new IllegalStateException("Os itens de um pedido processado nao podem mudar.");
        }
        validarItens(itens);
        this.itens = new ArrayList<>(itens);
    }

    /** Retorna a forma de pagamento recebida pelo pedido. */
    public FormaPagamento getFormaPagamento() {
        return formaPagamento;
    }

    /** Permite mudar a forma de pagamento somente enquanto o pedido esta pendente. */
    public void setFormaPagamento(FormaPagamento formaPagamento) {
        if (formaPagamento == null) {
            throw new IllegalArgumentException("A forma de pagamento e obrigatoria.");
        }
        if (status != null && status != StatusPedido.PENDENTE) {
            throw new IllegalStateException("O pagamento de um pedido processado nao pode mudar.");
        }
        this.formaPagamento = formaPagamento;
    }

    /** Retorna a etapa atual do pedido. */
    public StatusPedido getStatus() {
        return status;
    }

    /** Impede que o pedido seja criado ou alterado com uma lista vazia. */
    private void validarItens(ArrayList<ItemPedido> itens) {
        if (itens == null || itens.isEmpty() || itens.contains(null)) {
            throw new IllegalArgumentException("O pedido precisa de pelo menos um item valido.");
        }
    }
}

package Entidades;

/**
 * Dados e comportamentos comuns a comidas e bebidas.
 * E abstrata porque nao existe um produto "generico": todo produto e uma Comida ou uma Bebida.
 */
public abstract class Produto {
    private String nome;              // nome exibido no cardapio
    private Double preco;             // preco atual de venda, em reais
    private Integer quantidadeEstoque; // unidades disponiveis para venda
    private Integer codigo;           // identificador unico usado nas buscas

    /** Inicializa os atributos compartilhados pelas subclasses. */
    public Produto(String nome, Double preco, Integer quantidadeEstoque, Integer codigo) {
        this.nome = nome;
        this.preco = preco;
        this.quantidadeEstoque = quantidadeEstoque;
        this.codigo = codigo;
    }

    /** Retorna o nome exibido no cardapio. */
    public String getNome() {
        return nome;
    }

    /** Altera o nome do produto. */
    public void setNome(String nome) {
        this.nome = nome;
    }

    /** Retorna o preco atual do produto. */
    public Double getPreco() {
        return preco;
    }

    /** Altera o preco atual do produto. */
    public void setPreco(Double preco) {
        this.preco = preco;
    }

    /** Retorna a quantidade disponivel em estoque. */
    public Integer getQuantidadeEstoque() {
        return quantidadeEstoque;
    }

    /** Define a quantidade em estoque. */
    public void setQuantidadeEstoque(Integer quantidadeEstoque) {
        this.quantidadeEstoque = quantidadeEstoque;
    }

    /** Retorna o identificador do produto. */
    public Integer getCodigo() {
        return codigo;
    }

    /** Altera o identificador do produto. */
    public void setCodigo(Integer codigo) {
        this.codigo = codigo;
    }

    /** Acrescenta somente quantidades positivas ao estoque. */
    public void adicionarEstoque(int quantidade){
        if(quantidade > 0){
            quantidadeEstoque += quantidade;
        }
    }

    /** Retira unidades quando existe estoque suficiente. */
    public boolean removerEstoque(int quantidade){
        if(quantidade > 0 && quantidade <= quantidadeEstoque){
            quantidadeEstoque -= quantidade;
            return true;
        }
        return false;
    }

    /** Retorna o tipo do produto; cada subclasse informa o seu (polimorfismo). */
    public abstract String getTipo();

    /** Retorna o peso ou o volume ja formatado; cada subclasse sabe qual medida usa. */
    public abstract String getDetalhe();

    /**
     * Copia os dados comuns de outro produto para este mesmo objeto.
     * Alterar o objeto (em vez de troca-lo) mantem os pedidos ligados ao estoque atualizado.
     */
    public void atualizarDados(Produto novosDados) {
        setNome(novosDados.getNome());
        setPreco(novosDados.getPreco());
        setQuantidadeEstoque(novosDados.getQuantidadeEstoque());
    }

    /** Descreve os dados comuns; as subclasses acrescentam seus dados especificos. */
    @Override
    public String toString() {
        return  "Nome: " + nome + '\n' +
                "Preco: " + preco + '\n' +
                "Estoque: " + quantidadeEstoque + '\n' +
                "Codigo: " + codigo;
    }
}

package Gerenciamento;

import Entidades.Produto;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;

/** Guarda e gerencia os produtos disponiveis no Coffee Break. */
public class Cardapio {
    private final List<Produto> produtos; // estrutura de dados que guarda comidas e bebidas

    /** Recebe a estrutura de dados de fora para facilitar a injecao de dependencia. */
    public Cardapio(List<Produto> produtos) {
        if (produtos == null) {
            throw new IllegalArgumentException("A lista de produtos nao pode ser nula.");
        }
        this.produtos = produtos;
    }

    /** Inclui um produto valido, desde que seu codigo ainda nao exista. */
    public boolean adicionarProduto(Produto produto) {
        if (!produtoValido(produto) || buscarPorCodigo(produto.getCodigo()) != null) {
            return false;
        }
        return produtos.add(produto);
    }

    /** Encontra um produto pelo codigo ou retorna null. */
    public Produto buscarPorCodigo(int codigo) {
        for (Produto produto : produtos) {
            if (produto.getCodigo() != null && produto.getCodigo() == codigo) {
                return produto;
            }
        }
        return null;
    }

    /** Altera os dados de um produto, mantendo seu codigo e seu tipo original. */
    public boolean alterarProduto(int codigo, Produto novosDados) {
        if (!produtoValido(novosDados) || novosDados.getCodigo() != codigo) {
            return false;
        }
        Produto atual = buscarPorCodigo(codigo);
        // Uma comida so pode receber dados de comida, e uma bebida so de bebida.
        if (atual == null || atual.getClass() != novosDados.getClass()) {
            return false;
        }
        // Altera o mesmo objeto, para que os pedidos que ja usam o produto vejam o estoque novo.
        atual.atualizarDados(novosDados);
        return true;
    }

    /** Exclui um produto pelo codigo, se ele existir. */
    public boolean excluirProduto(int codigo) {
        Produto produto = buscarPorCodigo(codigo);
        return produto != null && produtos.remove(produto);
    }

    /** Retorna uma copia para impedir alteracoes externas na lista interna. */
    public List<Produto> listarProdutos() {
        return new ArrayList<>(produtos);
    }

    /** Devolve uma lista ordenada por nome, sem mudar a ordem armazenada. */
    public List<Produto> listarPorNome() {
        List<Produto> ordenados = listarProdutos();
        ordenados.sort(Comparator.comparing(Produto::getNome, String.CASE_INSENSITIVE_ORDER));
        return ordenados;
    }

    /** Devolve uma lista ordenada por preco, sem mudar a ordem armazenada. */
    public List<Produto> listarPorPreco() {
        List<Produto> ordenados = listarProdutos();
        ordenados.sort(Comparator.comparing(Produto::getPreco));
        return ordenados;
    }

    /** Exibe os produtos recebidos em uma tabela, inclusive listas ordenadas. */
    public void exibirProdutos(List<Produto> produtosParaExibir) {
        if (produtosParaExibir == null || produtosParaExibir.isEmpty()) {
            System.out.println("Nenhum produto cadastrado.");
            return;
        }
        // Larguras fixas no printf alinham as colunas da tabela.
        System.out.printf("%-6s %-8s %-22s %10s %8s %12s%n",
                "Codigo", "Tipo", "Nome", "Preco", "Estoque", "Peso/Volume");
        for (Produto produto : produtosParaExibir) {
            // Polimorfismo: getTipo() e getDetalhe() executam a versao de Comida ou de Bebida.
            System.out.printf("%-6d %-8s %-22.22s %10.2f %8d %12s%n",
                    produto.getCodigo(), produto.getTipo(), produto.getNome(), produto.getPreco(),
                    produto.getQuantidadeEstoque(), produto.getDetalhe());
        }
    }

    /** Confere os campos comuns antes de cadastrar ou alterar. */
    private boolean produtoValido(Produto produto) {
        return produto != null
                && produto.getCodigo() != null && produto.getCodigo() > 0
                && produto.getNome() != null && !produto.getNome().trim().isEmpty()
                && produto.getPreco() != null && Double.isFinite(produto.getPreco()) && produto.getPreco() > 0
                && produto.getQuantidadeEstoque() != null && produto.getQuantidadeEstoque() >= 0;
    }
}

package Pagamentos;

import Interfaces.FormaPagamento;

/** Registro manual de um pagamento com cartao de credito ou debito. */
public class PagamentoCartao implements FormaPagamento {
    /** Modalidades de cartao aceitas pelo Coffee Break. */
    public enum Tipo {
        CREDITO,
        DEBITO
    }

    private final Tipo tipo; // modalidade fixa deste objeto: credito ou debito

    /** Define a modalidade escolhida antes de associar o pagamento ao pedido. */
    public PagamentoCartao(Tipo tipo) {
        if (tipo == null) {
            throw new IllegalArgumentException("Escolha credito ou debito.");
        }
        this.tipo = tipo;
    }

    /** Valida o valor que o atendente esta registrando. */
    @Override
    public boolean pagar(double valor) {
        return Double.isFinite(valor) && valor > 0;
    }

    /** Informa no pedido se o cartao e de credito ou debito. */
    @Override
    public String getNome() {
        return tipo == Tipo.CREDITO ? "Cartao de credito" : "Cartao de debito";
    }
}

package Pagamentos;

import Interfaces.FormaPagamento;

/** Registro manual de um pagamento em Pix, sem conexao bancaria. */
public class PagamentoPix implements FormaPagamento {
    /** Aprova o registro quando o valor transferido por Pix e positivo. */
    @Override
    public boolean pagar(double valor) {
        return Double.isFinite(valor) && valor > 0;
    }

    /** Nome desta forma de pagamento nos menus e tabelas. */
    @Override
    public String getNome() {
        return "Pix";
    }
}

package Progama;

import Entidades.Bebida;
import Entidades.Comida;
import Entidades.ItemPedido;
import Entidades.Pedido;
import Entidades.Produto;
import Enums.StatusPedido;
import Gerenciamento.Cardapio;
import Gerenciamento.GerenciadorPedidos;
import Interfaces.FormaPagamento;
import Pagamentos.PagamentoCartao;
import Pagamentos.PagamentoDinheiro;
import Pagamentos.PagamentoPix;
import java.util.ArrayList;
import java.util.List;
import java.util.NoSuchElementException;
import java.util.Scanner;

/**
 * COFFEE BREAK - sistema textual de cadastro e gerenciamento de uma lanchonete.
 *
 * O que o programa faz:
 * - Cardapio: incluir, alterar, excluir, buscar por codigo, listar em tabela e
 *   listar ordenado por nome ou preco as comidas e bebidas.
 * - Pedidos: incluir com varios itens, alterar enquanto pendente, excluir, buscar,
 *   listar em tabela e ordenar por codigo ou total.
 * - Fluxo do pedido: PENDENTE -> PROCESSANDO (pagamento registrado e estoque baixado)
 *   -> CONCLUIDO; um pedido pendente tambem pode ser CANCELADO.
 * - Ao iniciar, cadastra 7 comidas, 7 bebidas e 7 pedidos para testes.
 * Os registros ficam em listas (ArrayList) na memoria enquanto o programa roda.
 *
 * Esta e a classe principal. Ela cria todas as dependencias e as entrega aos objetos
 * (inversao de controle) e conduz os menus pelo terminal.
 */
public class Programa {
    /** Monta os objetos, cadastra os dados iniciais e repete o menu principal ate o usuario sair. */
    public static void main(String[] args) {
        // Inversao de controle: o main cria as dependencias concretas e as injeta nos construtores.
        Cardapio cardapio = new Cardapio(new ArrayList<>());
        GerenciadorPedidos gerenciadorPedidos = new GerenciadorPedidos(new ArrayList<>());
        // Formas de pagamento disponiveis, todas guardadas pelo tipo da interface.
        List<FormaPagamento> pagamentos = new ArrayList<>();
        pagamentos.add(new PagamentoPix());
        pagamentos.add(new PagamentoDinheiro());
        pagamentos.add(new PagamentoCartao(PagamentoCartao.Tipo.CREDITO));
        pagamentos.add(new PagamentoCartao(PagamentoCartao.Tipo.DEBITO));

        // Objetos iniciais para exercitar as operacoes pedidas na AV3.
        cardapio.adicionarProduto(new Comida("Coxinha", 6.00, 20, 1, 120.0));
        cardapio.adicionarProduto(new Comida("Pao de queijo", 4.50, 25, 2, 70.0));
        cardapio.adicionarProduto(new Comida("Sanduiche", 12.00, 15, 3, 180.0));
        cardapio.adicionarProduto(new Comida("Brigadeiro", 3.50, 30, 4, 35.0));
        cardapio.adicionarProduto(new Comida("Bolo", 7.00, 12, 5, 100.0));
        cardapio.adicionarProduto(new Comida("Empada", 6.50, 18, 6, 110.0));
        cardapio.adicionarProduto(new Comida("Cookie", 5.00, 22, 7, 60.0));
        cardapio.adicionarProduto(new Bebida("Cafe expresso", 5.50, 30, 8, 50));
        cardapio.adicionarProduto(new Bebida("Cafe com leite", 7.00, 25, 9, 200));
        cardapio.adicionarProduto(new Bebida("Cha", 5.00, 20, 10, 200));
        cardapio.adicionarProduto(new Bebida("Suco de laranja", 8.00, 18, 11, 300));
        cardapio.adicionarProduto(new Bebida("Agua", 3.00, 40, 12, 500));
        cardapio.adicionarProduto(new Bebida("Refrigerante", 6.00, 24, 13, 350));
        cardapio.adicionarProduto(new Bebida("Chocolate quente", 9.00, 15, 14, 250));
        // Cria 7 pedidos: cada um tem a comida de codigo N e a bebida de codigo N + 7.
        for (int codigo = 1; codigo <= 7; codigo++) {
            ArrayList<ItemPedido> itens = new ArrayList<>();
            itens.add(new ItemPedido(cardapio.buscarPorCodigo(codigo), codigo % 3 + 1));
            itens.add(new ItemPedido(cardapio.buscarPorCodigo(codigo + 7), 1));
            // Alterna as formas de pagamento entre os pedidos iniciais.
            gerenciadorPedidos.adicionarPedido(
                    new Pedido(codigo, itens, pagamentos.get((codigo - 1) % pagamentos.size())));
        }

        // try-with-resources fecha o Scanner automaticamente ao sair.
        try (Scanner scanner = new Scanner(System.in)) {
            boolean aberto = true; // controla a repeticao do menu principal
            while (aberto) {
                System.out.println("\n========== COFFEE BREAK ==========");
                System.out.println("1 - Listar produtos");
                System.out.println("2 - Cadastrar produto");
                System.out.println("3 - Alterar produto");
                System.out.println("4 - Excluir produto");
                System.out.println("5 - Buscar produto por codigo");
                System.out.println("6 - Ordenar e listar produtos");
                System.out.println("7 - Gerenciar pedidos");
                System.out.println("0 - Sair");
                System.out.print("Opcao: ");
                if (!scanner.hasNextLine()) break; // entrada encerrada: sai do programa
                String opcao = scanner.nextLine().trim();

                // Erros de digitacao (ex.: letra no lugar de numero) sao tratados no catch abaixo.
                try {
                    switch (opcao) {
                        case "1": // Listar todos os produtos em tabela
                            cardapio.exibirProdutos(cardapio.listarProdutos());
                            break;
                        case "2": { // Incluir uma comida ou uma bebida
                            System.out.print("1 - Comida | 2 - Bebida: ");
                            int tipo = Integer.parseInt(scanner.nextLine().trim());
                            if (tipo != 1 && tipo != 2) {
                                System.out.println("Tipo invalido.");
                                break;
                            }
                            System.out.print("Codigo: ");
                            int codigo = Integer.parseInt(scanner.nextLine().trim());
                            // O codigo identifica o produto, entao nao pode repetir.
                            if (codigo <= 0 || cardapio.buscarPorCodigo(codigo) != null) {
                                System.out.println("O codigo deve ser positivo e ainda nao utilizado.");
                                break;
                            }
                            System.out.print("Nome: ");
                            String nome = scanner.nextLine().trim();
                            System.out.print("Preco: ");
                            // Aceita virgula ou ponto como separador decimal.
                            double preco = Double.parseDouble(scanner.nextLine().trim().replace(',', '.'));
                            System.out.print("Quantidade em estoque: ");
                            int estoque = Integer.parseInt(scanner.nextLine().trim());
                            Produto novo; // tipo da superclasse guarda Comida ou Bebida
                            if (tipo == 1) {
                                System.out.print("Peso em gramas: ");
                                double peso = Double.parseDouble(scanner.nextLine().trim().replace(',', '.'));
                                if (!Double.isFinite(peso) || peso <= 0) {
                                    System.out.println("O peso deve ser positivo.");
                                    break;
                                }
                                novo = new Comida(nome, preco, estoque, codigo, peso);
                            } else {
                                System.out.print("Volume em mL: ");
                                int volume = Integer.parseInt(scanner.nextLine().trim());
                                if (volume <= 0) {
                                    System.out.println("O volume deve ser positivo.");
                                    break;
                                }
                                novo = new Bebida(nome, preco, estoque, codigo, volume);
                            }
                            // O Cardapio valida nome, preco e estoque antes de incluir.
                            System.out.println(cardapio.adicionarProduto(novo)
                                    ? "Produto cadastrado." : "Dados do produto invalidos.");
                            break;
                        }
                        case "3": { // Alterar os dados de um produto existente
                            System.out.print("Codigo a alterar: ");
                            Produto atual = cardapio.buscarPorCodigo(
                                    Integer.parseInt(scanner.nextLine().trim()));
                            if (atual == null) {
                                System.out.println("Produto nao encontrado.");
                                break;
                            }
                            System.out.print("Novo nome: ");
                            String nome = scanner.nextLine().trim();
                            System.out.print("Novo preco: ");
                            double preco = Double.parseDouble(scanner.nextLine().trim().replace(',', '.'));
                            System.out.print("Novo estoque: ");
                            int estoque = Integer.parseInt(scanner.nextLine().trim());
                            // Objeto temporario com os novos dados; o tipo e o codigo continuam os mesmos.
                            Produto novosDados;
                            // O menu precisa saber o tipo para perguntar peso (comida) ou volume (bebida).
                            if (atual instanceof Comida) {
                                System.out.print("Novo peso em gramas: ");
                                double peso = Double.parseDouble(scanner.nextLine().trim().replace(',', '.'));
                                if (!Double.isFinite(peso) || peso <= 0) {
                                    System.out.println("O peso deve ser positivo.");
                                    break;
                                }
                                novosDados = new Comida(nome, preco, estoque, atual.getCodigo(), peso);
                            } else {
                                System.out.print("Novo volume em mL: ");
                                int volume = Integer.parseInt(scanner.nextLine().trim());
                                if (volume <= 0) {
                                    System.out.println("O volume deve ser positivo.");
                                    break;
                                }
                                novosDados = new Bebida(nome, preco, estoque, atual.getCodigo(), volume);
                            }
                            System.out.println(cardapio.alterarProduto(atual.getCodigo(), novosDados)
                                    ? "Produto alterado." : "Nao foi possivel alterar o produto.");
                            break;
                        }
                        case "4": { // Excluir um produto pelo codigo
                            System.out.print("Codigo a excluir: ");
                            int codigo = Integer.parseInt(scanner.nextLine().trim());
                            System.out.println(cardapio.excluirProduto(codigo)
                                    ? "Produto excluido." : "Produto nao encontrado.");
                            break;
                        }
                        case "5": { // Buscar um produto pelo atributo codigo
                            System.out.print("Codigo a buscar: ");
                            Produto produto = cardapio.buscarPorCodigo(
                                    Integer.parseInt(scanner.nextLine().trim()));
                            // A implementacao de toString depende da subclasse concreta.
                            System.out.println(produto == null ? "Produto nao encontrado." : produto);
                            break;
                        }
                        case "6": { // Listar produtos ordenados pelo criterio escolhido
                            System.out.print("1 - Nome | 2 - Preco: ");
                            String criterio = scanner.nextLine().trim();
                            if (criterio.equals("1")) cardapio.exibirProdutos(cardapio.listarPorNome());
                            else if (criterio.equals("2")) cardapio.exibirProdutos(cardapio.listarPorPreco());
                            else System.out.println("Criterio invalido.");
                            break;
                        }
                        case "7": { // Abrir o submenu de pedidos
                            boolean emPedidos = true; // controla a repeticao do submenu
                            while (emPedidos) {
                                System.out.println("\n------------ PEDIDOS ------------");
                                System.out.println("1 - Listar pedidos");
                                System.out.println("2 - Cadastrar pedido");
                                System.out.println("3 - Buscar pedido por codigo");
                                System.out.println("4 - Alterar pedido pendente");
                                System.out.println("5 - Excluir pedido");
                                System.out.println("6 - Ordenar e listar pedidos");
                                System.out.println("7 - Confirmar pagamento");
                                System.out.println("8 - Concluir pedido");
                                System.out.println("9 - Cancelar pedido");
                                System.out.println("0 - Voltar");
                                System.out.print("Opcao: ");
                                if (!scanner.hasNextLine()) {
                                    aberto = false;
                                    break;
                                }
                                String opcaoPedido = scanner.nextLine().trim();
                                try {
                                    switch (opcaoPedido) {
                                        case "1": // Listar todos os pedidos em tabela
                                            gerenciadorPedidos.exibirPedidos(gerenciadorPedidos.listarPedidos());
                                            break;
                                        // Incluir (2) e alterar (4) usam a mesma leitura de itens e pagamento.
                                        case "2":
                                        case "4": {
                                            boolean alterar = opcaoPedido.equals("4");
                                            System.out.print(alterar ? "Codigo a alterar: " : "Codigo do novo pedido: ");
                                            int codigo = Integer.parseInt(scanner.nextLine().trim());
                                            Pedido existente = gerenciadorPedidos.buscarPorCodigo(codigo);
                                            if (alterar && (existente == null
                                                    || existente.getStatus() != StatusPedido.PENDENTE)) {
                                                System.out.println("Pedido inexistente ou ja processado.");
                                                break;
                                            }
                                            if (!alterar && (codigo <= 0 || existente != null)) {
                                                System.out.println("O codigo deve ser positivo e ainda nao utilizado.");
                                                break;
                                            }
                                            ArrayList<ItemPedido> itens = new ArrayList<>();
                                            cardapio.exibirProdutos(cardapio.listarProdutos());
                                            // Le itens ate o usuario digitar 0.
                                            while (true) {
                                                System.out.print("Codigo do produto (0 para terminar): ");
                                                int codigoProduto = Integer.parseInt(scanner.nextLine().trim());
                                                if (codigoProduto == 0) break;
                                                Produto produto = cardapio.buscarPorCodigo(codigoProduto);
                                                if (produto == null) {
                                                    System.out.println("Produto nao encontrado.");
                                                    continue;
                                                }
                                                System.out.print("Quantidade: ");
                                                int quantidade = Integer.parseInt(scanner.nextLine().trim());
                                                if (quantidade <= 0) {
                                                    System.out.println("A quantidade deve ser positiva.");
                                                    continue;
                                                }
                                                // Considera tambem as unidades ja incluidas no mesmo pedido.
                                                int disponivel = produto.getQuantidadeEstoque();
                                                for (ItemPedido item : itens) {
                                                    if (item.getProduto() == produto) disponivel -= item.getQuantidade();
                                                }
                                                if (quantidade > disponivel) {
                                                    System.out.println("Quantidade disponivel para incluir: " + disponivel);
                                                    continue;
                                                }
                                                itens.add(new ItemPedido(produto, quantidade));
                                                System.out.println("Item incluido.");
                                            }
                                            if (itens.isEmpty()) {
                                                System.out.println(alterar
                                                        ? "Alteracao cancelada: o pedido precisa de itens."
                                                        : "Pedido nao criado: inclua pelo menos um item.");
                                                break;
                                            }
                                            // Repete a pergunta ate uma forma de pagamento valida ser escolhida.
                                            FormaPagamento pagamento = null;
                                            while (pagamento == null) {
                                                System.out.print("Forma de pagamento: 1 - Pix | 2 - Dinheiro | 3 - Cartao: ");
                                                String escolha = scanner.nextLine().trim();
                                                if (escolha.equals("1")) pagamento = pagamentos.get(0);
                                                else if (escolha.equals("2")) pagamento = pagamentos.get(1);
                                                else if (escolha.equals("3")) {
                                                    System.out.print("Cartao: 1 - Credito | 2 - Debito: ");
                                                    String tipoCartao = scanner.nextLine().trim();
                                                    if (tipoCartao.equals("1")) pagamento = pagamentos.get(2);
                                                    else if (tipoCartao.equals("2")) pagamento = pagamentos.get(3);
                                                    else System.out.println("Escolha 1 ou 2.");
                                                } else System.out.println("Escolha 1, 2 ou 3.");
                                            }
                                            if (alterar) {
                                                if (gerenciadorPedidos.alterarItens(codigo, itens)) {
                                                    existente.setFormaPagamento(pagamento);
                                                    System.out.println("Pedido alterado.");
                                                } else {
                                                    System.out.println("Nao foi possivel alterar o pedido.");
                                                }
                                            } else {
                                                // Inversao de controle: o menu escolhe a implementacao de pagamento.
                                                // Injecao de dependencia: o construtor recebe a FormaPagamento escolhida.
                                                Pedido pedido = new Pedido(codigo, itens, pagamento);
                                                System.out.println(gerenciadorPedidos.adicionarPedido(pedido)
                                                        ? String.format("Pedido cadastrado. Total: R$ %.2f",
                                                                pedido.calcularTotal())
                                                        : "Nao foi possivel cadastrar o pedido.");
                                            }
                                            break;
                                        }
                                        case "3": { // Buscar um pedido pelo codigo e mostrar seus itens
                                            System.out.print("Codigo a buscar: ");
                                            Pedido pedido = gerenciadorPedidos.buscarPorCodigo(
                                                    Integer.parseInt(scanner.nextLine().trim()));
                                            if (pedido == null) System.out.println("Pedido nao encontrado.");
                                            else pedido.exibirDetalhes();
                                            break;
                                        }
                                        case "5": { // Excluir um pedido pelo codigo
                                            System.out.print("Codigo a excluir: ");
                                            int codigo = Integer.parseInt(scanner.nextLine().trim());
                                            System.out.println(gerenciadorPedidos.excluirPedido(codigo)
                                                    ? "Pedido excluido." : "Pedido nao encontrado.");
                                            break;
                                        }
                                        case "6": { // Listar pedidos ordenados pelo criterio escolhido
                                            System.out.print("1 - Codigo | 2 - Total: ");
                                            String criterio = scanner.nextLine().trim();
                                            if (criterio.equals("1")) {
                                                gerenciadorPedidos.exibirPedidos(gerenciadorPedidos.listarPorCodigo());
                                            } else if (criterio.equals("2")) {
                                                gerenciadorPedidos.exibirPedidos(gerenciadorPedidos.listarPorTotal());
                                            } else System.out.println("Criterio invalido.");
                                            break;
                                        }
                                        // Mudancas de status: 7 paga, 8 conclui, 9 cancela.
                                        case "7":
                                        case "8":
                                        case "9": {
                                            System.out.print("Codigo do pedido: ");
                                            Pedido pedido = gerenciadorPedidos.buscarPorCodigo(
                                                    Integer.parseInt(scanner.nextLine().trim()));
                                            if (pedido == null) {
                                                System.out.println("Pedido nao encontrado.");
                                            } else if (opcaoPedido.equals("7")) {
                                                // Confirmar o pagamento tambem baixa o estoque dos produtos.
                                                System.out.println(pedido.confirmarPagamento()
                                                        ? "Pagamento registrado; pedido em processamento."
                                                        : "Pedido ja processado, sem estoque suficiente ou pagamento recusado.");
                                            } else if (opcaoPedido.equals("8")) {
                                                System.out.println(pedido.confirmarPedido()
                                                        ? "Pedido concluido."
                                                        : "O pedido precisa estar em processamento.");
                                            } else {
                                                System.out.println(pedido.cancelarPedido()
                                                        ? "Pedido cancelado."
                                                        : "Somente pedidos pendentes podem ser cancelados.");
                                            }
                                            break;
                                        }
                                        case "0": // Voltar ao menu principal
                                            emPedidos = false;
                                            break;
                                        default:
                                            System.out.println("Opcao invalida.");
                                    }
                                } catch (IllegalArgumentException erro) {
                                    System.out.println("Entrada invalida: " + erro.getMessage());
                                }
                            }
                            break;
                        }
                        case "0": // Sair do programa
                            System.out.println("Ate logo!");
                            aberto = false;
                            break;
                        default:
                            System.out.println("Opcao invalida.");
                    }
                } catch (IllegalArgumentException erro) {
                    System.out.println("Entrada invalida: " + erro.getMessage());
                } catch (NoSuchElementException fim) {
                    // Entrada fechada durante uma operacao: encerra o programa.
                    aberto = false;
                }
            }
        }
    }
}
